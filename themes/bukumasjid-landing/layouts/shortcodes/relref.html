{{- /* Shortcode relref — return URL murni (bukan tag ). Dipakai di dalam markdown link: [teks]({{< relref "blog/xxx.md" >}}) Fungsi template `relref` meresolve path relatif ke content ke URL. */ -}} {{- $path := .Get 0 | default "" -}} {{- $path = strings.TrimSpace $path -}} {{- $path = strings.Trim $path "\"" -}} {{- $clean := strings.TrimPrefix $path "/" -}} {{- if $clean -}} {{- relref .Page $clean -}} {{- else -}} {{- "/" -}} {{- end -}}
